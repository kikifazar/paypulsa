const PAYPULSA_API = "https://paypulsa-api.shanowellexcel.workers.dev"; async function buatPesanan() { const hasil = document.getElementById("hasil"); hasil.innerText = "Menghubungkan ke PayPal Sandbox..."; try { const res = await fetch( `${PAYPULSA_API}/api/create-order`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({}) } ); const data = await res.json(); if (!res.ok || !data.approval_url) { throw new Error( data.error || "Gagal membuat pesanan PayPal" ); } hasil.innerText = "Pesanan berhasil dibuat. Mengalihkan ke PayPal..."; window.location.href = data.approval_url; } catch (error) { hasil.innerText = "Terjadi kesalahan: " + error.message; } }
